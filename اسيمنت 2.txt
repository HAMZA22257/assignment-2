import 'dart:io';

void main() {
  stdout.write("enter n: ");
  int n = int.parse(stdin.readLineSync()!);
  int sum = 0;
  for (int i = 1; i <= n; i++) {
    stdout.write("$i ");
    sum = sum + i;
  }
  print("");
  print("sum = $sum");
}

Q4

import 'dart:io';

void main() {
  stdout.write("how many numbers? ");
  int count = int.parse(stdin.readLineSync()!);
  int pos = 0;
  int neg = 0;
  int zero = 0;
  for (int i = 1; i <= count; i++) {
    stdout.write("enter number $i: ");
    int x = int.parse(stdin.readLineSync()!);
    if (x > 0) {
      pos++;
    } else if (x < 0) {
      neg++;
    } else {
      zero++;
    }
  }
  print("positive = $pos");
  print("negative = $neg");
  print("zeros = $zero");
}

Q6

import 'dart:io';

void main() {
  stdout.write("enter number: ");
  int n = int.parse(stdin.readLineSync()!);
  int rev = 0;
  while (n > 0) {
    int digit = n % 10;
    rev = rev * 10 + digit;
    n = n ~/ 10;
  }
  print("reversed = $rev");
}

import 'dart:io';

void main() {
  stdout.write("how many numbers? ");
  int count = int.parse(stdin.readLineSync()!);
  stdout.write("enter number 1: ");
  int x = int.parse(stdin.readLineSync()!);
  int max = x;
  int min = x;
  for (int i = 2; i <= count; i++) {
    stdout.write("enter number $i: ");
    x = int.parse(stdin.readLineSync()!);
    if (x > max) {
      max = x;
    }
    if (x < min) {
      min = x;
    }
  }
  print("max = $max");
  print("min = $min");
}
